# MongoDB schema and setup for Receipt OCR + Expense Categorizer

# Users collection
# {
#   _id: ObjectId,
#   name: string,
#   email: string (unique),
#   password_hash: string,
#   created_at: datetime
# }

# Categories collection
# {
#   _id: ObjectId,
#   name: string (unique),
#   description: string (optional)
# }

# Receipts collection
# {
#   _id: ObjectId,
#   user_id: ObjectId (ref users),
#   image_uri: string,
#   ocr_text: string,
#   merchant_name: string,
#   purchase_date: date,
#   currency: string,
#   created_at: datetime
# }

# Expenses collection
# {
#   _id: ObjectId,
#   receipt_id: ObjectId (ref receipts),
#   category_id: ObjectId (ref categories),
#   amount: float,
#   tax_amount: float,
#   created_at: datetime
# }

# Optional: line_items collection
# {
#   _id: ObjectId,
#   receipt_id: ObjectId (ref receipts),
#   description: string,
#   quantity: float,
#   unit_price: float
# }

# Index suggestions:
# - users: { email: 1 }
# - categories: { name: 1 }
# - receipts: { user_id: 1, purchase_date: 1 }, { merchant_name: 1 }
# - expenses: { category_id: 1, created_at: 1 }
